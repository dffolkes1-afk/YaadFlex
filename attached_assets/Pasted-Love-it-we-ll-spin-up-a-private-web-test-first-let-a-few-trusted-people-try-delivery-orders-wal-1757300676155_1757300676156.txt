// middleware/basicAuth.js
export function basicAuth(req, res, next) {
  const auth = req.headers.authorization || "";
  const [type, b64] = auth.split(" ");
  if (type !== "Basic" || !b64) return ask();
  const [u, p] = Buffer.from(b64, "base64").toString().split(":");
  if (u === process.env.DEMO_USER && p === process.env.DEMO_PASS) return next();
  return ask();
  function ask() {
    res.set("WWW-Authenticate", 'Basic realm="YardFlex Demo"');
    res.status(401).send("Auth required");
  }
}

// server.js
import express from "express";
import { basicAuth } from "./middleware/basicAuth.js";
const app = express();
app.use("/demo", basicAuth); // protect /demo routes + UI

3) Money Flow (test mode — “the money part”)

Implement a mock YFX ledger + sandbox top-up:

Ledger rules

Store all values in YFX (1 YFX = 1 KWD).

Save FX snapshot on each tx (so the receipt shows JMD/USD equivalents).


Sample schema

-- users(id, name, phone)
-- wallets(user_id, balance_yfx DECIMAL(18,6))
-- tx(id, user_id, type, amount_yfx, fx_kwd_to_jmd, fx_kwd_to_usd, created_at, meta JSONB)
-- orders(id, buyer_id, seller_id, total_yfx, status, created_at)

Key endpoints

POST /api/wallet/topup-sandbox {amount_yfx}   // adds fake YFX for testing
POST /api/wallet/send         {to, amount_yfx, note}
POST /api/orders/create       {items:[{id,qty}], address} // holds YFX
POST /api/orders/complete     {order_id} // releases YFX to seller (85/15 split if service)
GET  /api/tx                  // list wallet transactions

Receipts show 2.4 YFX  ≈ $1,200 JMD  ≈ $7.80 USD (fx@2025-09-08)
