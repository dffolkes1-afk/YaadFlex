// ========================================== // server/routes.fuel.ts (Fuel Rewards API) // ========================================== import express from "express"; import crypto from "node:crypto"; import { db } from "../drizzle/db"; import { fuelPartners, fuelStations, fuelOffers, driverFuelAccounts, fuelTransactions, fuelRedemptions, fuelSettlements } from "../drizzle/schema.fuel"; import { and, eq, inArray, gte, lte } from "drizzle-orm";

const r = express.Router(); const uuid = () => crypto.randomUUID?.() || ${Date.now()};

// --- Admin: partners + stations + offers --- r.post("/api/fuel/partners", async (req, res) => { const [row] = await db.insert(fuelPartners).values({ id: uuid(), ...req.body }).returning(); res.json(row); });

r.post("/api/fuel/stations", async (req, res) => { const [row] = await db.insert(fuelStations).values({ id: uuid(), ...req.body }).returning(); res.json(row); });

r.post("/api/fuel/offers", async (req, res) => { const [row] = await db.insert(fuelOffers).values({ id: uuid(), ...req.body }).returning(); res.json(row); });

r.get("/api/fuel/offers/active", async (req, res) => { const now = new Date(); const rows = await db.select().from(fuelOffers).where(and(eq(fuelOffers.isActive, true), gte(fuelOffers.endsAt, now), lte(fuelOffers.startsAt, now))); res.json(rows); });

// --- Driver: enroll & balance --- r.post("/api/fuel/enroll", async (req, res) => { const { driverId, partnerId } = req.body; const [row] = await db.insert(driverFuelAccounts).values({ id: uuid(), driverId, partnerId, pointsBalance: "0", cashbackBalance: "0.00" }).returning(); res.json(row); });

r.get("/api/fuel/balance/:driverId", async (req, res) => { const rows = await db.select().from(driverFuelAccounts).where(eq(driverFuelAccounts.driverId, req.params.driverId)); res.json(rows); });

// --- Driver: record a purchase (from receipt or gateway callback) --- r.post("/api/fuel/purchase", async (req, res) => { const { driverId, partnerId, stationId, currency = "JMD", amount, liters, receiptRef } = req.body; const now = new Date(); const activeOffers = await db.select().from(fuelOffers).where(and(eq(fuelOffers.partnerId, partnerId), eq(fuelOffers.isActive, true), gte(fuelOffers.endsAt, now), lte(fuelOffers.startsAt, now)));

// pick best offer by highest reward value let best = activeOffers[0] || null; if (activeOffers.length > 1 && stationId) { best = activeOffers.find(o => !o.stationScope || (Array.isArray(o.stationScope) && o.stationScope.includes(stationId))) || activeOffers[0]; }

let points = "0" as any; let cashback = "0.00" as any; let offerId: string | null = null; if (best && Number(amount) >= Number(best.minSpend || 0)) { offerId = best.id; if (best.rewardType === "points" && best.pointsPerCurrency) { points = (Number(amount) * Number(best.pointsPerCurrency)).toFixed(2); } if (best.rewardType === "cashback" && best.cashbackRate) { cashback = (Number(amount) * Number(best.cashbackRate)).toFixed(2); } }

const [tx] = await db.insert(fuelTransactions).values({ id: uuid(), driverId, partnerId, stationId, currency, amount, liters, offerAppliedId: offerId || undefined, pointsEarned: points, cashbackEarned: cashback, receiptRef }).returning();

// update balances const accts = await db.select().from(driverFuelAccounts).where(and(eq(driverFuelAccounts.driverId, driverId), eq(driverFuelAccounts.partnerId, partnerId))); if (accts[0]) { const p = (Number(accts[0].pointsBalance) + Number(points)).toFixed(2); const c = (Number(accts[0].cashbackBalance) + Number(cashback)).toFixed(2); await db.update(driverFuelAccounts).set({ pointsBalance: p, cashbackBalance: c, updatedAt: new Date() }).where(eq(driverFuelAccounts.id, accts[0].id)); }

res.json({ tx, reward: { points, cashback, offerId } }); });

// --- Driver: redeem --- r.post("/api/fuel/redeem", async (req, res) => { const { driverId, partnerId, type, amount } = req.body; // type: points|cashback const [row] = await db.insert(fuelRedemptions).values({ id: uuid(), driverId, partnerId, type, amount, status: "pending" }).returning(); res.json(row); });

// --- Admin: approve redemption & run settlement --- r.post("/api/fuel/redemptions/:id/approve", async (req, res) => { const id = req.params.id; const now = new Date(); await db.update(fuelRedemptions).set({ status: "approved", approvedAt: now }).where(eq(fuelRedemptions.id, id)); res.json({ ok: true }); });

r.post("/api/fuel/settlements/run", async (req, res) => { const { partnerId, period } = req.body; // e.g. 2025-09-week36 // Sum approved cashback + issued points in period // (For simplicity, we sum from fuelTransactions over timeframe in real impl.) const [settlement] = await db.insert(fuelSettlements).values({ id: uuid(), partnerId, period, totalCashback: "0.00", totalPoints: "0", status: "open" }).returning(); res.json(settlement); });

export default r;
