5) Paste-in: capture console errors to your API

Creates a paper trail when testers hit bugs.

<script>
  (function(){
    const post = (data) => fetch('/api/logs/frontend', {
      method:'POST', headers:{'Content-Type':'application/json'},
      body: JSON.stringify({ ts: Date.now(), ua: navigator.userAgent, ...data })
    }).catch(()=>{});
    window.addEventListener('error', e => post({ type:'error', msg: e.message, stack: e.error?.stack || '', src: e.filename, line: e.lineno }));
    window.addEventListener('unhandledrejection', e => post({ type:'promise', msg: (e.reason && e.reason.message)||String(e.reason), stack: e.reason && e.reason.stack }));
    console.log = new Proxy(console.log, { apply(t, thisArg, args){ post({ type:'log', msg: args.join(' ') }); return Reflect.apply(t, thisArg, args); }});
  })();
</script>

6) Paste-in: dark map style (Leaflet quick start)

If you’re not using a map yet, here’s a simple dark style:

<link rel="stylesheet" href="https://unpkg.com/leaflet@1.9.4/dist/leaflet.css"/>
<div id="map" style="height:60vh;border-radius:16px;overflow:hidden"></div>
<script src="https://unpkg.com/leaflet@1.9.4/dist/leaflet.js"></script>
<script>
  const map = L.map('map', { zoomControl:false }).setView([18.0179,-76.8099], 12); // Kingston
  L.tileLayer('https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png', { attribution:'©OpenStreetMap, ©CARTO' }).addTo(map);
  // sample driver pins
  const drivers = [
    {name:'Driver A', lat:18.012, lng:-76.79},
    {name:'Driver B', lat:18.04, lng:-76.82},
  ];
  drivers.forEach(d => L.marker([d.lat,d.lng]).addTo(map).bindPopup(`<b>${d.name}</b><br>Online`));
</script>
