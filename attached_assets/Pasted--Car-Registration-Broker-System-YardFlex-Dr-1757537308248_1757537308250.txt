// ============================================== // drizzle/schema.broker.ts // ============================================== import { pgTable, varchar, text, boolean, timestamp, integer, decimal, jsonb } from "drizzle-orm/pg-core";

export const brokers = pgTable("brokers", { id: varchar("id", { length: 36 }).primaryKey(), fullName: varchar("fullName", { length: 160 }).notNull(), email: varchar("email", { length: 160 }).notNull(), phone: varchar("phone", { length: 40 }), govId: varchar("govId", { length: 64 }), // TRN or national ID verified: boolean("verified").default(false), rating: decimal("rating", { precision: 3, scale: 2 }), isActive: boolean("isActive").default(true), createdAt: timestamp("createdAt").defaultNow(), updatedAt: timestamp("updatedAt").defaultNow() });

export const brokerRegions = pgTable("broker_regions", { id: varchar("id", { length: 36 }).primaryKey(), brokerId: varchar("brokerId", { length: 36 }).notNull(), parish: varchar("parish", { length: 64 }).notNull(), // e.g. "Kingston", "St. Andrew" active: boolean("active").default(true), createdAt: timestamp("createdAt").defaultNow() });

export const regServiceCatalog = pgTable("reg_service_catalog", { code: varchar("code", { length: 32 }).primaryKey(), // 'NEW_REG', 'RENEWAL', 'TRANSFER', 'RED_PLATE', 'REPLACEMENT' name: varchar("name", { length: 120 }).notNull(), baseFeeJMD: integer("baseFeeJMD").notNull(), // your service fee (not TAJ fee) estGovFeeJMD: integer("estGovFeeJMD").default(0), // estimated TAJ/Transport Authority fees (informational) slaHours: integer("slaHours").default(72), active: boolean("active").default(true) });

export const regRequests = pgTable("reg_requests", { id: varchar("id", { length: 36 }).primaryKey(), applicationId: varchar("applicationId", { length: 36 }).notNull(), // FK -> registrationApplications.id ownerId: varchar("ownerId", { length: 36 }).notNull(), vehicleId: varchar("vehicleId", { length: 36 }).notNull(), serviceCode: varchar("serviceCode", { length: 32 }).notNull(), // FK -> regServiceCatalog.code parish: varchar("parish", { length: 64 }).notNull(), pickupAddress: text("pickupAddress"), deliveryAddress: text("deliveryAddress"), notes: text("notes"), quotedFeeJMD: integer("quotedFeeJMD"), govFeeJMD: integer("govFeeJMD"), status: varchar("status", { length: 32 }).default("pending_quote"), createdAt: timestamp("createdAt").defaultNow(), updatedAt: timestamp("updatedAt").defaultNow() });

export const regEvents = pgTable("reg_events", { id: varchar("id", { length: 36 }).primaryKey(), requestId: varchar("requestId", { length: 36 }).notNull(), type: varchar("type", { length: 40 }).notNull(), // 'QUOTE_CREATED','ASSIGNED','PICKED_UP','SUBMITTED_TAJ','APPROVED','REJECTED','DELIVERED' meta: jsonb("meta"), createdAt: timestamp("createdAt").defaultNow() });

export const regAssignments = pgTable("reg_assignments", { id: varchar("id", { length: 36 }).primaryKey(), requestId: varchar("requestId", { length: 36 }).notNull(), brokerId: varchar("brokerId", { length: 36 }).notNull(), status: varchar("status", { length: 24 }).default("assigned"), // 'assigned'|'in_progress'|'completed'|'cancelled' pickupAt: timestamp("pickupAt"), deliveredAt: timestamp("deliveredAt"), createdAt: timestamp("createdAt").defaultNow(), updatedAt: timestamp("updatedAt").defaultNow() });

export const regPayouts = pgTable("reg_payouts", { id: varchar("id", { length: 36 }).primaryKey(), brokerId: varchar("brokerId", { length: 36 }).notNull(), requestId: varchar("requestId", { length: 36 }).notNull(), amountJMD: integer("amountJMD").notNull(), status: varchar("status", { length: 24 }).default("pending"), // 'pending'|'paid'|'failed' provider: varchar("provider", { length: 24 }).default("stripe_connect"), providerPayoutId: varchar("providerPayoutId", { length: 80 }), createdAt: timestamp("createdAt").defaultNow() });

export const regSLA = pgTable("reg_sla", { id: varchar("id", { length: 36 }).primaryKey(), requestId: varchar("requestId", { length: 36 }).notNull(), startAt: timestamp("startAt").defaultNow(), dueAt: timestamp("dueAt"), breached: boolean("breached").default(false) });

// ============================================== // server/routes.broker.ts // ============================================== import express from "express"; import crypto from "node:crypto"; import { and, eq, desc } from "drizzle-orm"; import { db } from "../drizzle/db"; import { brokers, brokerRegions, regServiceCatalog, regRequests, regEvents, regAssignments, regPayouts, regSLA } from "../drizzle/schema.broker"; import { registrationApplications, vehicles } from "../drizzle/schema.vehicle";

const router = express.Router(); const uid = () => crypto.randomUUID();

// ---- Pricing/Commission policy (can move to DB/config) ---- const COMMISSION = { brokerShare: 0.6, // 60% of YardFlex service fee goes to broker yardflexShare: 0.4 };

// ---- Create or update broker ---- router.post("/api/brokers", async (req, res) => { const id = req.body.id || uid(); const [row] = await db.insert(brokers).values({ id, fullName: req.body.fullName, email: req.body.email, phone: req.body.phone, govId: req.body.govId, verified: !!req.body.verified }).onConflictDoUpdate({ target: brokers.id, set: { fullName: req.body.fullName, email: req.body.email, phone: req.body.phone, govId: req.body.govId, verified: !!req.body.verified, updatedAt: new Date() } }).returning(); res.json(row); });

// ---- Broker regions ---- router.post("/api/brokers/:brokerId/regions", async (req, res) => { const [row] = await db.insert(brokerRegions).values({ id: uid(), brokerId: req.params.brokerId, parish: req.body.parish, active: true }).returning(); res.json(row); });

// ---- Service catalog seed/add ---- router.post("/api/reg/services", async (req, res) => { const [row] = await db.insert(regServiceCatalog).values({ code: req.body.code, name: req.body.name, baseFeeJMD: req.body.baseFeeJMD, estGovFeeJMD: req.body.estGovFeeJMD || 0, slaHours: req.body.slaHours || 72, active: true }).returning(); res.json(row); });

router.get("/api/reg/services", async (_req, res) => { const rows = await db.select().from(regServiceCatalog); res.json(rows); });

// ---- Create registration request (driver submits) ---- router.post("/api/reg/requests", async (req, res) => { const id = uid(); // simple quote from catalog const [svc] = await db.select().from(regServiceCatalog).where(eq(regServiceCatalog.code, req.body.serviceCode)); if (!svc) return res.status(400).json({ error: "Unknown serviceCode" }); const quoted = Number(svc.baseFeeJMD);

const [row] = await db.insert(regRequests).values({ id, applicationId: req.body.applicationId, ownerId: req.body.ownerId, vehicleId: req.body.vehicleId, serviceCode: req.body.serviceCode, parish: req.body.parish, pickupAddress: req.body.pickupAddress, deliveryAddress: req.body.deliveryAddress, notes: req.body.notes, quotedFeeJMD: quoted, govFeeJMD: svc.estGovFeeJMD, status: "pending_payment" }).returning();

await db.insert(regEvents).values({ id: uid(), requestId: id, type: "QUOTE_CREATED", meta: { quoted } }); res.json(row); });

// ---- List open requests (for dispatcher UI) ---- router.get("/api/reg/requests", async (req, res) => { const parish = req.query.parish as string | undefined; let rows = await db.select().from(regRequests); if (parish) rows = rows.filter(r => r.parish === parish); rows.sort((a,b)=> b.createdAt.getTime() - a.createdAt.getTime()); res.json(rows); });

// ---- Assign to a broker ---- router.post("/api/reg/requests/:id/assign", async (req, res) => { const requestId = req.params.id; const brokerId = req.body.brokerId; const [assign] = await db.insert(regAssignments).values({ id: uid(), requestId, brokerId, status: "assigned" }).returning(); await db.insert(regEvents).values({ id: uid(), requestId, type: "ASSIGNED", meta: { brokerId } }); res.json(assign); });

// ---- Update assignment status ---- router.post("/api/reg/assignments/:id/status", async (req, res) => { const [row] = await db.update(regAssignments) .set({ status: req.body.status, pickupAt: req.body.pickupAt || undefined, deliveredAt: req.body.deliveredAt || undefined, updatedAt: new Date() }) .where(eq(regAssignments.id, req.params.id)) .returning(); await db.insert(regEvents).values({ id: uid(), requestId: row.requestId, type: ASSIGN_${req.body.status.toUpperCase()} }); res.json(row); });

// ---- Mark request state (dispatcher/admin) ---- router.post("/api/reg/requests/:id/status", async (req, res) => { const [row] = await db.update(regRequests) .set({ status: req.body.status, updatedAt: new Date() }) .where(eq(regRequests.id, req.params.id)) .returning(); await db.insert(regEvents).values({ id: uid(), requestId: req.params.id, type: req.body.status.toUpperCase() }); res.json(row); });

// ---- Auto payout on completion ---- router.post("/api/reg/requests/:id/complete", async (req, res) => { const requestId = req.params.id; // mark delivered const [reqRow] = await db.update(regRequests) .set({ status: "delivered", updatedAt: new Date() }) .where(eq(regRequests.id, requestId)) .returning();

// find assignment const assigns = await db.select().from(regAssignments).where(eq(regAssignments.requestId, requestId)); const assign = assigns[0]; if (!assign) return res.status(400).json({ error: "No assignment" });

const brokerAmount = Math.round(Number(reqRow.quotedFeeJMD || 0) * COMMISSION.brokerShare); const [payout] = await db.insert(regPayouts).values({ id: uid(), requestId, brokerId: assign.brokerId, amountJMD: brokerAmount, status: "pending" }).returning();

await db.insert(regEvents).values({ id: uid(), requestId, type: "DELIVERED", meta: { brokerAmount } }); res.json({ request: reqRow, payout }); });

// ---- Payout status webhook (e.g., Stripe Connect transfer) ---- router.post("/webhooks/payouts", async (req, res) => { // Validate signature separately… then update payout const providerPayoutId = req.body.providerPayoutId; const status = req.body.status; // 'paid'|'failed' const [row] = await db.update(regPayouts) .set({ status, providerPayoutId: providerPayoutId }) .where(eq(regPayouts.id, req.body.payoutId)) .returning(); res.json(row); });

export default router;

// ============================================== // server/index.ts (wire it) // ============================================== import express from "express"; import brokerRoutes from "./routes.broker";

const app = express(); app.use(express.json()); app.use(brokerRoutes);

const port = process.env.PORT || 4000; app.listen(port, () => console.log(Broker API running on http://localhost:${port}));
