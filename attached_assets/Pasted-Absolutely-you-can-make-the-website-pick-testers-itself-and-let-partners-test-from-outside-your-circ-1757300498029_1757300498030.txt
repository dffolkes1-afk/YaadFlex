🔐 Gate middleware (Express – drop-in)

// auth/gate.js
import crypto from "crypto";
export async function gate(req, res, next) {
  // 1) Allow admin by session
  if (req.user?.role === "admin") return next();

  // 2) Magic link or access token
  const t = req.query.t || req.headers["x-ydx"];
  if (!t) return deny(res);

  const invite = await db.invites.findOne({ token: t });
  if (!invite) return deny(res);

  if (new Date(invite.expires_at) < new Date()) return deny(res, "expired");

  // 3) Bind session to tester & mark as used (if one-time)
  req.tester = await db.testers.findOne({ id: invite.tester_id });
  if (!req.session.tester_id) {
    req.session.tester_id = invite.tester_id;
    if (!invite.used_at) await db.invites.update(invite.id, { used_at: new Date() });
  }
  return next();
}
function deny(res, why="unauthorized") {
  res.status(401).json({ ok:false, error: why });
}
export function newToken() {
  return crypto.randomBytes(24).toString("base64url");
}

Use it on beta routes:

app.use("/beta", gate, express.static("dist-beta"));

🧪 Cohort selection logic (auto-pick testers)

Waitlist form captures: email, region, device, use-case (driver/vendor/rider), connection speed.

Auto-scoring example:

Region = “JM” (+3), device = “Android low-end” (+2), role = “driver” (+3)

Score ≥ 6 → auto-invite, else keep on waitlist.



// invite/selector.js
export function score(candidate) {
  let s=0;
  if (candidate.region==="JM") s+=3;
  if (candidate.device?.includes("Android")) s+=2;
  if (candidate.role==="driver") s+=3;
  return s;
}
export async function maybeInvite(c) {
  const s = score(c);
  if (s >= 6) {
    const token = newToken();
    await db.invites.insert({ tester_id: c.id, token, expires_at: addHours(48) });
    sendMagicLink(c.email, `https://yourapp.com/beta?t=${token}`);
    await db.testers.update(c.id, { role: "invited" });
  }
}

Flag check helper:

export async function canSee(tester, flagKey) {
  const f = await db.feature_flags.findOne({ key: flagKey });
  if (!f || !f.enabled) return false;
  if (f.audience==="all") return true;
  if (!tester) return false;
  if (f.audience==="beta_tester" && ["beta_tester","admin"].includes(tester.role)) return true;
  if (f.audience==="partner_tester" && ["partner_tester","admin"].includes(tester.role)) return true;
  return false;
}
