import React, { useMemo, useRef, useState } from "react"; import { Canvas, useFrame } from "@react-three/fiber"; import { OrbitControls, Stars, Html } from "@react-three/drei";

/**

YaadFlex World Emblem – React Three.js Mockup (R3F)


---

What it shows:

Spinning globe with glow


Orbiting "people" silhouettes holding hands (abstracted)


Duo silhouettes above globe (inspired vibe, original shapes)


UI toggles for speed and glow


How to run (Replit / Vite):

1. npm i react react-dom three @react-three/fiber @react-three/drei



2. Place this file as App.jsx (or a route component) and render <App/> */




function GlowRing({ radius = 3.05, thickness = 0.03, color = "#1cb0f6" }) { const ref = useRef(); useFrame((_, d) => { if (ref.current) ref.current.rotation.y += d * 0.15; }); return ( <mesh ref={ref}> <torusGeometry args={[radius, thickness, 32, 128]} /> <meshStandardMaterial emissive={color} emissiveIntensity={0.7} color="#0b1d35" metalness={0.2} roughness={0.3} /> </mesh> ); }

function Globe({ speed = 0.4 }) { const ref = useRef(); useFrame((_, d) => { if (ref.current) ref.current.rotation.y += d * speed; }); return ( <group> <mesh ref={ref}> <sphereGeometry args={[3, 64, 64]} /> {/* Replace with a texture if desired using <meshStandardMaterial map={texture} /> */} <meshStandardMaterial color="#0e3a5b" emissive="#0a6cab" emissiveIntensity={0.25} metalness={0.15} roughness={0.6} /> </mesh> <GlowRing /> </group> ); }

function Person({ scale = 0.18, color = "#000000" }) { // Abstract silhouette: sphere head + capsule-like body return ( <group scale={scale}> {/* Body /} <mesh position={[0, 0, 0]}> <capsuleGeometry args={[0.6, 1.8, 8, 16]} /> <meshStandardMaterial color={color} roughness={0.8} metalness={0.05} /> </mesh> {/ Head /} <mesh position={[0, 1.6, 0]}> <sphereGeometry args={[0.55, 16, 16]} /> <meshStandardMaterial color={color} roughness={0.8} metalness={0.05} /> </mesh> {/ Simple arm link to suggest holding hands */} <mesh position={[0, 0.4, 0]} rotation={[0, 0, Math.PI / 2]}> <cylinderGeometry args={[0.12, 0.12, 2.2, 8]} /> <meshStandardMaterial color={color} roughness={0.8} metalness={0.05} /> </mesh> </group> ); }

function PeopleOrbit({ count = 8, radius = 3.4, speed = 0.6, glow = "#ffd700" }) { const group = useRef(); const angles = useMemo(() => new Array(count).fill(0).map((, i) => (i / count) * Math.PI * 2), [count]); useFrame((, d) => { if (group.current) group.current.rotation.y += d * speed; }); return ( <group ref={group}> {angles.map((a, i) => ( <group key={i} rotation={[0, a, 0]} position={[0, 0, 0]}> <group position={[Math.sin(a) * radius, 0, Math.cos(a) * radius]} rotation={[0, -a + Math.PI, 0]}> <Person /> {/* tiny glow dot to hint hand-holding connection */} <mesh position={[0.2, 0.4, 0]}> <sphereGeometry args={[0.04, 8, 8]} /> <meshStandardMaterial emissive={glow} emissiveIntensity={1.2} color={glow} /> </mesh> </group> </group> ))} </group> ); }

function DuoSilhouettes() { // Brain-style: shorter, stockier; Pinky-style: taller, slimmer return ( <group position={[0, 3.9, 0]}> {/* Brain-style /} <group position={[-0.6, 0, 0]} scale={0.6}> <mesh position={[0, 0.5, 0]}> <capsuleGeometry args={[0.7, 1.2, 8, 16]} /> <meshStandardMaterial color="#111111" metalness={0.1} roughness={0.8} /> </mesh> <mesh position={[0, 1.7, 0]}> <sphereGeometry args={[0.6, 16, 16]} /> <meshStandardMaterial color="#111111" /> </mesh> {/ arms folded hint */} <mesh position={[0, 1.0, 0]} rotation={[0, 0, Math.PI / 2]}> <cylinderGeometry args={[0.12, 0.12, 1.2, 8]} /> <meshStandardMaterial color="#111111" /> </mesh> </group>

{/* Pinky-style */}
  <group position={[0.6, 0, 0]} scale={0.7}>
    <mesh position={[0, 0.8, 0]}>
      <capsuleGeometry args={[0.5, 1.8, 8, 16]} />
      <meshStandardMaterial color="#191919" />
    </mesh>
    <mesh position={[0, 2.1, 0]}>
      <sphereGeometry args={[0.5, 16, 16]} />
      <meshStandardMaterial color="#191919" />
    </mesh>
    {/* slight slouch: rotate whole group */}
  </group>

  {/* Accent glow */}
  <mesh position={[0, -0.1, 0]}>
    <ringGeometry args={[1.5, 1.8, 64]} />
    <meshBasicMaterial color="#e83e8c" opacity={0.4} transparent />
  </mesh>
</group>

); }

function Scene({ globeSpeed, orbitSpeed }) { return ( <> <ambientLight intensity={0.6} /> <pointLight position={[6, 6, 6]} intensity={1.2} /> <Stars radius={60} depth={40} count={8000} factor={2} fade speed={0.4} /> <Globe speed={globeSpeed} /> <PeopleOrbit speed={orbitSpeed} /> <DuoSilhouettes /> <OrbitControls enablePan={false} /> </> ); }

export default function App() { const [globeSpeed, setGlobeSpeed] = useState(0.4); const [orbitSpeed, setOrbitSpeed] = useState(0.6);

return ( <div className="min-h-screen w-full bg-gradient-to-b from-black to-[#0b1d35] text-white"> <div className="max-w-5xl mx-auto p-4 sm:p-6"> <header className="flex items-center justify-between mb-4"> <h1 className="text-xl sm:text-2xl font-semibold tracking-tight">YaadFlex World Emblem – Live Mock</h1> <div className="text-xs opacity-70">R3F demo • spin + orbit</div> </header>

<div className="rounded-2xl overflow-hidden shadow-2xl ring-1 ring-white/10">
      <Canvas camera={{ position: [0, 4.2, 8], fov: 50 }}>
        <Scene globeSpeed={globeSpeed} orbitSpeed={orbitSpeed} />
        <Html position={[0, -4.2, 0]} center>
          <div className="text-center text-white/70 text-xs">Drag to orbit • Scroll to zoom</div>
        </Html>
      </Canvas>
    </div>

    <div className="mt-4 grid grid-cols-1 sm:grid-cols-2 gap-3">
      <div className="bg-white/5 rounded-xl p-4 flex items-center justify-between">
        <label className="text-sm">Globe speed</label>
        <input
          type="range"
          min={0}
          max={1}
          step={0.01}
          value={globeSpeed}
          onChange={(e) => setGlobeSpeed(parseFloat(e.target.value))}
          className="w-40"
        />
      </div>
      <div className="bg-white/5 rounded-xl p-4 flex items-center justify-between">
        <label className="text-sm">Orbit speed</label>
        <input
          type="range"
          min={0}
          max={1.2}
          step={0.01}
          value={orbitSpeed}
          onChange={(e) => setOrbitSpeed(parseFloat(e.target.value))}
          className="w-40"
        />
      </div>
    </div>

    <p className="mt-4 text-sm text-white/70">Tip: Replace the globe material with a texture for higher realism, or add country color themes. This mock uses original abstract silhouettes (no copyrighted characters).</p>
  </div>
</div>

); }
