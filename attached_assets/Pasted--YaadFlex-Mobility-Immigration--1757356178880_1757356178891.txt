// =============================== // File: src/db/schema.ts // =============================== import { pgTable, uuid, varchar, boolean, timestamp, integer, jsonb, numeric } from "drizzle-orm/pg-core";

export const hotels = pgTable("hotels", { id: uuid("id").primaryKey().defaultRandom(), name: varchar("name", { length: 200 }).notNull(), email: varchar("email", { length: 200 }).notNull(), phone: varchar("phone", { length: 50 }), address: varchar("address", { length: 300 }), country: varchar("country", { length: 2 }).notNull(), starRating: integer("star_rating").default(3), isActive: boolean("is_active").default(true), createdAt: timestamp("created_at").defaultNow(), updatedAt: timestamp("updated_at").defaultNow() });

export const hotelRoles = pgTable("hotel_roles", { id: uuid("id").primaryKey().defaultRandom(), hotelId: uuid("hotel_id").notNull().references(() => hotels.id), title: varchar("title", { length: 120 }).notNull(), dept: varchar("dept", { length: 120 }), minSkill: varchar("min_skill", { length: 40 }), payRate: numeric("pay_rate", { precision: 10, scale: 2 }), payType: varchar("pay_type", { length: 20 }), notes: varchar("notes", { length: 500 }), isActive: boolean("is_active").default(true) });

export const hotelShifts = pgTable("hotel_shifts", { id: uuid("id").primaryKey().defaultRandom(), hotelId: uuid("hotel_id").notNull().references(() => hotels.id), roleId: uuid("role_id").notNull().references(() => hotelRoles.id), start: timestamp("start").notNull(), end: timestamp("end").notNull(), slots: integer("slots").default(1), notes: varchar("notes", { length: 500 }), status: varchar("status", { length: 20 }).default("open") });

export const immigrationPrograms = pgTable("immigration_programs", { id: uuid("id").primaryKey().defaultRandom(), name: varchar("name", { length: 200 }).notNull(), country: varchar("country", { length: 2 }).notNull(), visaType: varchar("visa_type", { length: 60 }).notNull(), seasonal: boolean("seasonal").default(false), requirements: jsonb("requirements"), // string[] documentsRequired: jsonb("documents_required"), // string[] feeUsd: numeric("fee_usd", { precision: 10, scale: 2 }).default("0"), status: varchar("status", { length: 20 }).default("open") });

export const immigrationApplications = pgTable("immigration_applications", { id: uuid("id").primaryKey().defaultRandom(), programId: uuid("program_id").notNull().references(() => immigrationPrograms.id), workerId: uuid("worker_id").notNull(), status: varchar("status", { length: 20 }).default("pending"), submittedAt: timestamp("submitted_at").defaultNow(), decisionAt: timestamp("decision_at"), assignedLawyerId: uuid("assigned_lawyer_id"), notes: varchar("notes", { length: 500 }) });

export const legalProviders = pgTable("legal_providers", { id: uuid("id").primaryKey().defaultRandom(), firmName: varchar("firm_name", { length: 200 }).notNull(), contactEmail: varchar("contact_email", { length: 200 }).notNull(), barId: varchar("bar_id", { length: 120 }), country: varchar("country", { length: 2 }).notNull(), specialties: jsonb("specialties"), // string[] rating: numeric("rating", { precision: 2, scale: 1 }).default("0"), isActive: boolean("is_active").default(true) });

export const cases = pgTable("cases", { id: uuid("id").primaryKey().defaultRandom(), userId: uuid("user_id").notNull(), lawyerId: uuid("lawyer_id").notNull().references(() => legalProviders.id), type: varchar("type", { length: 60 }).notNull(), status: varchar("status", { length: 20 }).default("open"), openedAt: timestamp("opened_at").defaultNow(), closedAt: timestamp("closed_at"), documents: jsonb("documents"), // string[] of doc IDs messagesThreadId: uuid("messages_thread_id") });

export const kycProfiles = pgTable("kyc_profiles", { id: uuid("id").primaryKey().defaultRandom(), userId: uuid("user_id").notNull().unique(), tier: varchar("tier", { length: 20 }).default("tier1"), status: varchar("status", { length: 20 }).default("pending"), pepScreen: boolean("pep_screen").default(false), sanctionsHit: boolean("sanctions_hit").default(false), lastReviewedAt: timestamp("last_reviewed_at") });

export const documents = pgTable("documents", { id: uuid("id").primaryKey().defaultRandom(), ownerId: uuid("owner_id").notNull(), type: varchar("type", { length: 60 }).notNull(), path: varchar("path", { length: 400 }).notNull(), hash: varchar("hash", { length: 128 }), issuedBy: varchar("issued_by", { length: 200 }), expiresAt: timestamp("expires_at"), verified: boolean("verified").default(false), notes: varchar("notes", { length: 500 }) });

export const travelItineraries = pgTable("travel_itineraries", { id: uuid("id").primaryKey().defaultRandom(), userId: uuid("user_id").notNull(), pnr: varchar("pnr", { length: 12 }), segments: jsonb("segments"), price: numeric("price", { precision: 12, scale: 2 }), currency: varchar("currency", { length: 3 }).default("USD"), status: varchar("status", { length: 20 }).default("hold"), provider: varchar("provider", { length: 60 }), ticketedAt: timestamp("ticketed_at") });

export const escrowAccounts = pgTable("escrow_accounts", { id: uuid("id").primaryKey().defaultRandom(), orderId: uuid("order_id").notNull(), type: varchar("type", { length: 40 }).notNull(), amount: numeric("amount", { precision: 12, scale: 2 }).notNull(), currency: varchar("currency", { length: 3 }).default("USD"), status: varchar("status", { length: 20 }).default("open"), releasedAt: timestamp("released_at"), refundedAt: timestamp("refunded_at") });

// =============================== // File: src/db/index.ts // =============================== import { drizzle } from "drizzle-orm/node-postgres"; import { Pool } from "pg"; import * as schema from "./schema"; import dotenv from "dotenv";

dotenv.config(); const pool = new Pool({ connectionString: process.env.DATABASE_URL }); export const db = drizzle(pool, { schema }); export type DB = typeof db;

// =============================== // File: src/validators.ts // =============================== import { z } from "zod";

export const createShiftSchema = z.object({ hotelId: z.string().uuid(), roleId: z.string().uuid(), start: z.string().datetime(), end: z.string().datetime(), slots: z.number().int().positive().default(1), notes: z.string().max(500).optional() });

export const immigrationAppSchema = z.object({ programId: z.string().uuid(), workerId: z.string().uuid(), docs: z.array(z.string()).default([]) });

export const travelSearchSchema = z.object({ from: z.string().min(3), to: z.string().min(3), depart: z.string(), return: z.string().optional(), pax: z.number().int().positive().default(1), cabin: z.enum(["ECONOMY","PREMIUM","BUSINESS","FIRST"]).default("ECONOMY") });

// =============================== // File: src/routes/hotelWorkforce.ts // =============================== import { Router } from "express"; import { db } from "../db"; import { hotels, hotelRoles, hotelShifts } from "../db/schema"; import { eq } from "drizzle-orm"; import { createShiftSchema } from "../validators";

const r = Router();

r.post("/hotels", async (req, res) => { const body = req.body || {}; const [row] = await db.insert(hotels).values({ name: body.name, email: body.email, phone: body.phone, address: body.address, country: body.country, }).returning(); res.json(row); });

r.get("/hotels", async (_req, res) => { const rows = await db.select().from(hotels); res.json(rows); });

r.post("/hotels/:id/roles", async (req, res) => { const { id } = req.params; const [row] = await db.insert(hotelRoles).values({ hotelId: id, ...req.body }).returning(); res.json(row); });

r.get("/hotels/:id/roles", async (req, res) => { const { id } = req.params; const rows = await db.select().from(hotelRoles).where(eq(hotelRoles.hotelId, id)); res.json(rows); });

r.post("/hotels/:id/shifts", async (req, res) => { const parsed = createShiftSchema.safeParse({ ...req.body, hotelId: req.params.id }); if (!parsed.success) return res.status(400).json(parsed.error.flatten()); const [row] = await db.insert(hotelShifts).values({ ...parsed.data }).returning(); res.json(row); });

r.get("/hotels/:id/shifts", async (req, res) => { const { id } = req.params; const rows = await db.select().from(hotelShifts).where(eq(hotelShifts.hotelId, id)); res.json(rows); });

r.post("/shifts/:id/assign", async (req, res) => { // Placeholder: implement worker assignment join table if needed res.json({ ok: true, shiftId: req.params.id, assigned: req.body.workerId }); });

r.post("/shifts/:id/checkin", async (req, res) => { res.json({ ok: true, shiftId: req.params.id, at: new Date().toISOString(), action: "checkin" }); });

r.post("/shifts/:id/checkout", async (req, res) => { res.json({ ok: true, shiftId: req.params.id, at: new Date().toISOString(), action: "checkout" }); });

export default r;

// =============================== // File: src/routes/immigration.ts // =============================== import { Router } from "express"; import { db } from "../db"; import { immigrationPrograms, immigrationApplications, legalProviders } from "../db/schema"; import { eq } from "drizzle-orm"; import { immigrationAppSchema } from "../validators";

const r2 = Router();

r2.get("/immigration/programs", async (_req, res) => { const rows = await db.select().from(immigrationPrograms); res.json(rows); });

r2.post("/immigration/applications", async (req, res) => { const parsed = immigrationAppSchema.safeParse(req.body); if (!parsed.success) return res.status(400).json(parsed.error.flatten()); const [row] = await db.insert(immigrationApplications).values({ programId: parsed.data.programId, workerId: parsed.data.workerId, notes: (parsed.data.docs || []).join(", ") }).returning(); res.json(row); });

r2.get("/immigration/applications/:id", async (req, res) => { const { id } = req.params; const [row] = await db.select().from(immigrationApplications).where(eq(immigrationApplications.id, id)); if (!row) return res.status(404).json({ error: "not_found" }); res.json(row); });

r2.post("/immigration/applications/:id/assign-lawyer", async (req, res) => { const { id } = req.params; const [row] = await db.update(immigrationApplications) .set({ assignedLawyerId: req.body.lawyerId }) .where(eq(immigrationApplications.id, id)) .returning(); res.json(row); });

r2.post("/immigration/applications/:id/decision", async (req, res) => { const { id } = req.params; const [row] = await db.update(immigrationApplications) .set({ status: req.body.status, decisionAt: new Date() }) .where(eq(immigrationApplications.id, id)).returning(); res.json(row); });

export default r2;
